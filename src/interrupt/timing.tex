% Timing diagram of the CPU's interrupt request interface: one hardware
% interrupt taken to completion, followed by a second request that arrives while
% the ISR is still running and is therefore made to wait.
%
% The interface is AXI-stream shaped -- irq_valid_i / irq_ready_o / irq_addr_i --
% and deliberately knows nothing about daisy chaining, which is a matter for the
% adaptation layer around the CPU.  See src/interrupt/README.md.
%
% This diagram is the SPECIFICATION of the protocol, not a recording of it --
% src/interrupt/interrupt.vhd (task T2) does not exist yet.  Redraw it against
% a real GHDL simulation once it does, exactly as src/cpu_main/timing.tex is
% read off test/prog_waveform.asm.  See doc/interrupts.md for the prose.

\documentclass{standalone}
\usepackage{timing}

% ACCEPTED lasts two cycles and does not fit in a default-width box.
\renewcommand{\wavescale}{1.7}

\begin{document}
% Drawing convention, the same one src/cpu_main/timing.tex uses: handshake bits
% are always drawn with their true value; a bus is greyed out ("x") in every
% cycle in which its owner is not driving it.
%
% Every signal below is either a pin (the top three) or a register output, with
% one deliberate exception: start_i is combinational.  Unlike the daisy-chain
% protocol this replaced, irq_addr_i is valid from the cycle the device asserts
% irq_valid_i and is held until the request is accepted, so the CPU never has to
% capture it inside a one-cycle window.
%
% The transfer happens in the single cycle t=4, where irq_valid_i and
% irq_ready_o are both high.  The CPU has already committed at t=3 and has
% already redirected FETCH there, from the copy of the address it registered at
% t=2; the accept is what releases the device, not what delivers the address.
\begin{wave}{11}{12}
 \nextwave{irq\_valid\_i}             \bit{0}{1} \bit{1}{4} \bit{0}{2} \bit{1}{5}
 \nextwave{irq\_addr\_i}              \unknown[x]{1} \known{ISR}{4} \unknown[x]{2} \known{ISR2}{5}
 \nextwave{irq\_ready\_o}             \bit{0}{4} \bit{1}{1} \bit{0}{7}
 \nextwave{interrupt/state}           \known{IDLE}{4} \known{ACCEPTED}{2} \known{IDLE}{6}
 \nextwave{interrupt/pending\_o}      \bit{0}{2} \bit{1}{2} \bit{0}{4} \bit{1}{4}
 \nextwave{interrupt/addr\_o}         \unknown[x]{2} \known{ISR}{6} \known{ISR2}{4}
 \nextwave{WRITE/start\_i}            \bit{0}{3} \bit{1}{1} \bit{0}{8}
 \nextwave{WRITE/inst\_done\_o}       \bit{0}{3} \bit{1}{1} \bit{0}{4} \bit{1}{1} \bit{0}{1} \bit{1}{1} \bit{0}{1}
 \nextwave{WRITE/int\_active}         \bit{0}{4} \bit{1}{7} \bit{0}{1}
 \nextwave{WRITE/fetch\_valid\_o}     \bit{0}{3} \bit{1}{1} \bit{0}{6} \bit{1}{1} \bit{0}{1}
 \nextwave{WRITE/fetch\_addr\_o}      \unknown[x]{3} \known{ISR}{1} \unknown[x]{6} \known{PC}{1} \unknown[x]{1}
\end{wave}

\end{document}
